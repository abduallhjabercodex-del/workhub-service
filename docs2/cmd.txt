-------------------------------------------------------------------------------
# 2.1 إنشاء مشروع جديد (POST Project)

  -H "Content-Type: application/json" \
  -d '{"name": "WorkHub Demo Project"}'

# 2.2 عرض المشاريع مع الترتيب والبحث والتصفح (GET Projects)

# 2.3 إنشاء مهمة جديدة لمشروع معين (POST Task)
curl -i -X POST http://127.0.0.1:8080/api/v1/projects/1/tasks \
  -H "Content-Type: application/json" \
  -d '{"title": "Design Database Schema and Setup Transactions"}'

# 2.4 عرض المهام مع الفلترة والصفحات (GET Tasks Filtering & Pagination)

# 2.5 جلب مهمة واحدة برقم المعرف (GET Single Task)
curl -i http://127.0.0.1:8080/api/v1/tasks/1

# 2.6 التحديث الجزئي لمهمة (PATCH Task)
curl -i -X PATCH http://127.0.0.1:8080/api/v1/tasks/1 \
  -H "Content-Type: application/json" \
  -d '{"status": "in_progress"}'

# 2.7 الحذف الناعم لمهمة (DELETE Task - Soft Delete)
curl -i -X DELETE http://127.0.0.1:8080/api/v1/tasks/1

# 2.8 إحصائيات المهام التجميعية لكل المشاريع
curl -i http://127.0.0.1:8080/api/v1/tasks/stats

# 2.9 إحصائيات المهام لمشروع معين
curl -i http://127.0.0.1:8080/api/v1/projects/1/tasks/stats

# 2.10 محاكاة خطأ التحقق واختبار غلاف الأخطاء الموحد (Validation Error Envelope)
